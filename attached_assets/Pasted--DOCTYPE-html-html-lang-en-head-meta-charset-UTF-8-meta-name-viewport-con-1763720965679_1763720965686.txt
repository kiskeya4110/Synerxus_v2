        // Step navigation functions
        function showStep(step) {
            // Hide all steps
            document.querySelectorAll('.step-content').forEach(content => {
                content.classList.remove('active');
            });
            
            // Show current step
            document.getElementById(`step${step}`).classList.add('active');
            
            // Update step indicators
            document.querySelectorAll('.step-indicator').forEach((indicator, index) => {
                indicator.classList.remove('active', 'completed');
                if (index + 1 < step) {
                    indicator.classList.add('completed');
                } else if (index + 1 === step) {
                    indicator.classList.add('active');
                }
            });
            
            // Update navigation buttons
            document.getElementById('prevBtn').style.display = step === 1 ? 'none' : 'block';
            document.getElementById('nextBtn').style.display = step === totalSteps ? 'none' : 'block';
            document.getElementById('submitBtn').style.display = step === totalSteps ? 'block' : 'none';
        }

            if (validateCurrentStep()) {
                if (currentStep < totalSteps) {

        function previousStep() {
            if (currentStep > 1) {

                showStep(currentStep);
            }
        }

        function validateCurrentStep() {
            const currentStepElement = document.getElementById(`step${currentStep}`);
            const requiredFields = currentStepElement.querySelectorAll('[required]');
            let isValid = true;

            requiredFields.forEach(field => {
                if (!field.checkValidity()) {
                    field.classList.add('border-red-500');
                    isValid = false;
                } else {
                    field.classList.remove('border-red-500');
                }
            });

            // Custom validations for specific steps
            if (currentStep === 1) {
                const focusAreas = document.querySelectorAll('input[name="focusAreas"]:checked');
                if (focusAreas.length === 0 || focusAreas.length > 5) {
                    showNotification('Please select 1-5 focus areas', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 2) {
                if (selectedSDGs.size === 0) {
                    showNotification('Please select at least one SDG', 'error');
                    isValid = false;
                }
                
                const sdgImpact = document.getElementById('sdgImpact');
                if (sdgImpact.value.length < 300) {
                    showNotification('SDG impact description must be at least 300 characters', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 4) {
                const languages = document.querySelectorAll('input[name="languages"]:checked');
                if (languages.length === 0 || languages.length > 3) {
                    showNotification('Please select 1-3 operating languages', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 5) {
                const skills = document.querySelectorAll('input[name="neededSkills"]:checked');
                if (skills.length === 0 || skills.length > 10) {
                    showNotification('Please select 1-10 needed skills', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 7) {
                const commMethods = document.querySelectorAll('input[name="commMethods"]:checked');
                if (commMethods.length === 0) {
                    showNotification('Please select at least one communication method', 'error');
                    isValid = false;
                }

                const experiencePrefs = document.querySelectorAll('input[name="experiencePrefs"]:checked');
                if (experiencePrefs.length === 0) {
                    showNotification('Please select at least one experience level preference', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 8) {
                const volunteerDesc = document.getElementById('volunteerDescription');
                const uniqueValue = document.getElementById('uniqueValue');
                
                if (volunteerDesc.value.length < 200) {
                    showNotification('Volunteer description must be at least 200 characters', 'error');
                    isValid = false;
                }
                
                if (uniqueValue.value.length < 150) {
                    showNotification('Unique value proposition must be at least 150 characters', 'error');
                    isValid = false;
                }
            }

            if (currentStep === 9) {
                const associations = document.querySelectorAll('input[name="associations"]:checked');
                if (associations.length === 0) {
                    showNotification('Please select at least one professional association or network', 'error');
                    isValid = false;
                }
            }

            if (!isValid) {
                showNotification('Please fill in all required fields correctly', 'error');
            }

            return isValid;
        }
